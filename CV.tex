\section{\makebox[\widthof{\faUsers}][c]{\color{CVBlue}\faUsers}\ 项目经历}


\textbf{基于 CARLA 的雨夜行人横穿避让场景仿真} \hfill 课程项目 \titlebreak
\begin{itemize}[nosep, topsep=\intraProjectListTopSep]
    \item \textbf{场景复现：} 构建夜间强降雨、前车急刹和行人横穿叠加的危险工况，配置湿滑路面、雾灯与固定地图点位，提升实验可复现性。
    \item \textbf{感知与控制：} 使用语义 LiDAR 进行黑盒风险识别，加入 0.25 s 延迟、0.75 m 测距噪声和 12\% 丢帧概率；依据距离和相对速度触发“紧急左变道--回正--制动”策略。
    \item \textbf{量化结果：} 最小 TTC 为 0.01 s，最大左移 4.09 m，最终航向误差 2.33°，仿真中实现无碰撞避险。
\end{itemize}

\textbf{基于水力测功器的发动机台架测控系统设计} \hfill 课程项目 \titlebreak
\begin{itemize}[nosep, topsep=\intraProjectListTopSep]
    \item \textbf{系统方案：} 规划“发动机--联轴器--水力测功器--测控柜--上位机”总体结构，完成转速、扭矩、油门、水温、油压和进排气等参数的传感器选型与接口设计。
    \item \textbf{硬件与通信：} 以 \textbf{STM32F407} 和 16 位 ADC 为核心规划多路采集、PWM/DO 输出及 HMI；设计 500 kbit/s CAN 主通道和 RS-485/Modbus 备用通道，并考虑心跳、超时和安全联锁。
    \item \textbf{控制策略：} 设计前馈加位置式 \textbf{PID} 的定扭矩控制方案，分析积分限幅、微分滤波、输出限幅和速率限制，并给出按转速分区整定的 Simulink 建模方案。
\end{itemize}


\projectsep


\textbf{进气道湿壁油膜建模与前馈补偿} \hfill 课程项目 \titlebreak
\begin{itemize}[nosep, topsep=\intraProjectListTopSep]
    \item \textbf{模型分析：} 基于油膜流量关系建立湿壁模型，使用 \textbf{Simulink} 比较附壁系数 \(X\) 和时间常数 \(\tau\) 对燃油动态传输的影响。
    \item \textbf{补偿设计：} 设计前馈补偿器，仿真中喷油指令峰值约 2.85，随后指数衰减至稳态 1，补偿后入缸燃油基本贴合目标需求。
    \item \textbf{工程反思：} 指出固定 \(X\)、\(\tau\) 在大节气门或不同水温下会产生过补偿或欠补偿，提出基于压力、转速、壁温等参数查表修正的自适应补偿方向。
\end{itemize}
